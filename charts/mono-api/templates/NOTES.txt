{{- else if .Values.ingress.enabled }}

  kubectl -n {{ .Release.Namespace }} get ingress {{ $fullname }}
{{- end }}
{{- else }}

  http://{{ $fullname }}.{{ .Release.Namespace }}.svc:{{ .Values.service.port }}/api/users
